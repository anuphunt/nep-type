cmake_minimum_required(VERSION 3.20)
# Ship the pad without requiring a separately installed Visual C++ runtime.
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(NepTypeWindowsPrototype LANGUAGES CXX)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_compile_definitions(NOMINMAX)
add_library(NepTypeComposition STATIC composition.cpp candidate_adapter.cpp)
target_include_directories(NepTypeComposition PUBLIC include)
target_compile_features(NepTypeComposition PUBLIC cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(NepTypeSetupTransactionTests tests/setup_transaction_test.cpp)
    target_include_directories(NepTypeSetupTransactionTests PRIVATE installer)
    target_compile_features(NepTypeSetupTransactionTests PRIVATE cxx_std_17)
    add_test(NAME NepTypeSetupTransactionTests COMMAND NepTypeSetupTransactionTests)
    add_executable(NepTypeKeyboardSessionTests tests/keyboard_session_test.cpp)
    target_link_libraries(NepTypeKeyboardSessionTests PRIVATE NepTypeComposition ${CMAKE_DL_LIBS})
    add_test(NAME NepTypeKeyboardSessionTests COMMAND NepTypeKeyboardSessionTests)
    add_executable(NepTypeLocalStateTests ../../shared/tests/local_experience_test.cpp)
    target_compile_features(NepTypeLocalStateTests PRIVATE cxx_std_17)
    add_test(NAME NepTypeLocalStateTests COMMAND NepTypeLocalStateTests)
    add_executable(NepTypePadCandidateViewTests tests/pad_candidate_view_test.cpp)
    target_include_directories(NepTypePadCandidateViewTests PRIVATE include)
    target_compile_features(NepTypePadCandidateViewTests PRIVATE cxx_std_17)
    add_test(NAME NepTypePadCandidateViewTests COMMAND NepTypePadCandidateViewTests)
    add_executable(NepTypeCompositionTests tests/composition_test.cpp)
    target_link_libraries(NepTypeCompositionTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeCompositionTests COMMAND NepTypeCompositionTests)
    add_executable(NepTypeCandidateTests tests/candidate_adapter_test.cpp)
    target_link_libraries(NepTypeCandidateTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeCandidateTests COMMAND NepTypeCandidateTests)
    add_executable(NepTypeTsfLifecycleTests tsf/service.cpp tsf/edit_bridge.cpp tests/tsf_lifecycle_test.cpp)
    target_compile_features(NepTypeTsfLifecycleTests PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeTsfLifecycleTests PRIVATE NEPTYPE_TSF_TEST)
    target_include_directories(NepTypeTsfLifecycleTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(NepTypeTsfLifecycleTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeTsfLifecycleTests COMMAND NepTypeTsfLifecycleTests)
    add_executable(NepTypeTsfEditTests tsf/edit_bridge.cpp tests/tsf_edit_test.cpp)
    target_compile_features(NepTypeTsfEditTests PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeTsfEditTests PRIVATE NEPTYPE_TSF_TEST)
    target_include_directories(NepTypeTsfEditTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(NepTypeTsfEditTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeTsfEditTests COMMAND NepTypeTsfEditTests)
    add_executable(NepTypeOwnedRangeTests tsf/owned_range.cpp tests/owned_range_test.cpp)
    target_compile_features(NepTypeOwnedRangeTests PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeOwnedRangeTests PRIVATE NEPTYPE_TSF_TEST)
    target_include_directories(NepTypeOwnedRangeTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(NepTypeOwnedRangeTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeOwnedRangeTests COMMAND NepTypeOwnedRangeTests)
    # These assertion-based host fixtures must also execute in Release builds.
    foreach(TEST_TARGET NepTypeSetupTransactionTests NepTypeKeyboardSessionTests NepTypeLocalStateTests NepTypePadCandidateViewTests NepTypeCompositionTests NepTypeCandidateTests NepTypeTsfLifecycleTests NepTypeTsfEditTests NepTypeOwnedRangeTests)
        if(MSVC)
            target_compile_options(${TEST_TARGET} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${TEST_TARGET} PRIVATE -UNDEBUG)

if(WIN32)
    enable_language(RC)
    add_executable(NepTypeWindows WIN32 main.cpp neptype.rc)
    target_include_directories(NepTypeWindows PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(NepTypeWindows PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(NepTypeWindows PRIVATE /utf-8)
    endif()
    target_link_libraries(NepTypeWindows PRIVATE user32 gdi32 comctl32 comdlg32 shell32 advapi32 NepTypeComposition)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(NepTypeWindows PRIVATE Threads::Threads)
    set(NEPTYPE_ENGINE_BINARY_DIR "" CACHE PATH "Directory containing NepTypeEngine.dll and its embedded dictionary")
    if(NOT EXISTS "${NEPTYPE_ENGINE_BINARY_DIR}/NepTypeEngine.dll")
        message(FATAL_ERROR "Build the shared Rust DLL with cargo build --release and set NEPTYPE_ENGINE_BINARY_DIR")
    endif()
    file(GLOB NEPTYPE_ENGINE_DLLS "${NEPTYPE_ENGINE_BINARY_DIR}/*.dll")
    foreach(ENGINE_DLL IN LISTS NEPTYPE_ENGINE_DLLS)
        add_custom_command(TARGET NepTypeWindows POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${ENGINE_DLL}" "$<TARGET_FILE_DIR:NepTypeWindows>")
    endforeach()
    add_custom_command(TARGET NepTypeWindows POST_BUILD
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/verify-engine-library.py" "$<TARGET_FILE_DIR:NepTypeWindows>/NepTypeEngine.dll"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/engine-data.py"
            stage "$<TARGET_FILE_DIR:NepTypeWindows>/dictionaries")

    if(MINGW)
        target_link_options(NepTypeWindows PRIVATE -municode)
    endif()
    if(BUILD_TESTING)
        add_executable(NepTypePadShortcutTests tests/pad_shortcuts_test.cpp neptype.rc)
        target_include_directories(NepTypePadShortcutTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_features(NepTypePadShortcutTests PRIVATE cxx_std_17)
        target_link_libraries(NepTypePadShortcutTests PRIVATE user32 gdi32 comctl32 comdlg32 shell32 advapi32 Threads::Threads)
        # The pad target stages the real Rust DLL and dictionaries beside both executables.
        add_dependencies(NepTypePadShortcutTests NepTypeWindows)
        if(MSVC)
            target_compile_options(NepTypePadShortcutTests PRIVATE /UNDEBUG)
        else()
            target_compile_options(NepTypePadShortcutTests PRIVATE -UNDEBUG)
        endif()
        add_test(NAME NepTypePadShortcutTests COMMAND NepTypePadShortcutTests)
        set_tests_properties(NepTypePadShortcutTests PROPERTIES TIMEOUT 60)
    endif()
    # The live keyboard is separate from the retained lifecycle fixture/pad.
    add_library(NepTypeTsf SHARED tsf/live_service.cpp tsf/registration.cpp tsf/owned_range.cpp tsf/input_scope_guids.cpp tsf/exports.def neptype.rc)
    target_include_directories(NepTypeTsf PRIVATE tsf include)
    target_compile_features(NepTypeTsf PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeTsf PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(NepTypeTsf PRIVATE NepTypeComposition ole32 oleaut32 uuid user32 gdi32 advapi32)
    add_dependencies(NepTypeTsf NepTypeWindows)
    option(NEPTYPE_TSF_DIAGNOSTICS "Emit activation diagnostics for native debugging" OFF)
    if(NEPTYPE_TSF_DIAGNOSTICS)
        target_compile_definitions(NepTypeTsf PRIVATE NEPTYPE_TSF_DIAGNOSTICS)
    endif()
    set(KEYBOARD_PAYLOAD_HEADER "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/keyboard_payload.h")
    add_custom_command(OUTPUT "${KEYBOARD_PAYLOAD_HEADER}"
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/installer/bind_payload.py"
            "$<TARGET_FILE:NepTypeTsf>" "$<TARGET_FILE_DIR:NepTypeTsf>/NepTypeEngine.dll" "${KEYBOARD_PAYLOAD_HEADER}"
        DEPENDS NepTypeTsf "${CMAKE_CURRENT_SOURCE_DIR}/installer/bind_payload.py" VERBATIM)
    add_executable(NepTypeKeyboardSetup installer/keyboard_setup.cpp tsf/registration.cpp "${KEYBOARD_PAYLOAD_HEADER}")
    target_compile_features(NepTypeKeyboardSetup PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeKeyboardSetup PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_include_directories(NepTypeKeyboardSetup PRIVATE tsf installer "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>")
    target_link_libraries(NepTypeKeyboardSetup PRIVATE ole32 oleaut32 uuid user32 shell32 advapi32 bcrypt)
    if(MINGW)
        target_link_options(NepTypeKeyboardSetup PRIVATE -municode)
    endif()
    if(BUILD_TESTING)
        add_test(NAME NepTypeKeyboardRealEngineTests COMMAND NepTypeKeyboardSessionTests "$<TARGET_FILE_DIR:NepTypeTsf>/NepTypeEngine.dll")
        add_executable(NepTypeNativeTsfTests tests/native_tsf_test.cpp tsf/input_scope_guids.cpp)
        target_compile_features(NepTypeNativeTsfTests PRIVATE cxx_std_17)
        target_compile_definitions(NepTypeNativeTsfTests PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
        target_include_directories(NepTypeNativeTsfTests PRIVATE tsf)
        target_link_libraries(NepTypeNativeTsfTests PRIVATE ole32 oleaut32 uuid user32 advapi32)
        if(MINGW)
            target_link_options(NepTypeNativeTsfTests PRIVATE -municode)
        endif()
        if(MSVC)
            target_compile_options(NepTypeNativeTsfTests PRIVATE /UNDEBUG)
        else()
            target_compile_options(NepTypeNativeTsfTests PRIVATE -UNDEBUG)
        endif()
        add_dependencies(NepTypeNativeTsfTests NepTypeTsf)
        add_test(NAME NepTypeNativeTsfTests COMMAND NepTypeNativeTsfTests "$<TARGET_FILE:NepTypeTsf>")
        set_tests_properties(NepTypeNativeTsfTests PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
    endif()
    install(TARGETS NepTypeTsf NepTypeWindows NepTypeKeyboardSetup RUNTIME DESTINATION . LIBRARY DESTINATION .)
    install(FILES "${NEPTYPE_ENGINE_BINARY_DIR}/NepTypeEngine.dll" install-keyboard.ps1 README.md DESTINATION .)
    install(DIRECTORY "$<TARGET_FILE_DIR:NepTypeWindows>/dictionaries" DESTINATION .)
    add_custom_command(TARGET NepTypeTsf POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/install-keyboard.ps1" "$<TARGET_FILE_DIR:NepTypeTsf>/install-keyboard.ps1")
endif()
