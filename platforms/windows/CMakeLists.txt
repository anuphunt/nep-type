cmake_minimum_required(VERSION 3.20)
# Ship the pad without requiring a separately installed Visual C++ runtime.
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(NepTypeWindowsPrototype LANGUAGES CXX)
if(MSVC)
    add_compile_options(/utf-8)
endif()
add_compile_definitions(NOMINMAX)
add_library(NepTypeComposition STATIC composition.cpp candidate_adapter.cpp)
target_include_directories(NepTypeComposition PUBLIC include)
target_compile_features(NepTypeComposition PUBLIC cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(NepTypeCompositionTests tests/composition_test.cpp)
    target_link_libraries(NepTypeCompositionTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeCompositionTests COMMAND NepTypeCompositionTests)
    add_executable(NepTypeCandidateTests tests/candidate_adapter_test.cpp)
    target_link_libraries(NepTypeCandidateTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeCandidateTests COMMAND NepTypeCandidateTests)
    add_executable(NepTypeTsfLifecycleTests tsf/service.cpp tsf/edit_bridge.cpp tests/tsf_lifecycle_test.cpp)
    target_compile_features(NepTypeTsfLifecycleTests PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeTsfLifecycleTests PRIVATE NEPTYPE_TSF_TEST)
    target_include_directories(NepTypeTsfLifecycleTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(NepTypeTsfLifecycleTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeTsfLifecycleTests COMMAND NepTypeTsfLifecycleTests)
    add_executable(NepTypeTsfEditTests tsf/edit_bridge.cpp tests/tsf_edit_test.cpp)
    target_compile_features(NepTypeTsfEditTests PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeTsfEditTests PRIVATE NEPTYPE_TSF_TEST)
    target_include_directories(NepTypeTsfEditTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(NepTypeTsfEditTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeTsfEditTests COMMAND NepTypeTsfEditTests)
    add_executable(NepTypeOwnedRangeTests tsf/owned_range.cpp tests/owned_range_test.cpp)
    target_compile_features(NepTypeOwnedRangeTests PRIVATE cxx_std_17)
    target_compile_definitions(NepTypeOwnedRangeTests PRIVATE NEPTYPE_TSF_TEST)
    target_include_directories(NepTypeOwnedRangeTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(NepTypeOwnedRangeTests PRIVATE NepTypeComposition)
    add_test(NAME NepTypeOwnedRangeTests COMMAND NepTypeOwnedRangeTests)
    # These assertion-based host fixtures must also execute in Release builds.
    foreach(TEST_TARGET NepTypeCompositionTests NepTypeCandidateTests NepTypeTsfLifecycleTests NepTypeTsfEditTests NepTypeOwnedRangeTests)
        if(MSVC)
            target_compile_options(${TEST_TARGET} PRIVATE /UNDEBUG)
        else()
            target_compile_options(${TEST_TARGET} PRIVATE -UNDEBUG)
        endif()
    endforeach()
endif()
if(WIN32)
    enable_language(RC)
    add_executable(NepTypeWindows WIN32 main.cpp neptype.rc)
    target_include_directories(NepTypeWindows PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_features(NepTypeWindows PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(NepTypeWindows PRIVATE /utf-8)
    endif()
    target_link_libraries(NepTypeWindows PRIVATE user32 gdi32 comctl32 NepTypeComposition)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    find_package(Threads REQUIRED)
    target_link_libraries(NepTypeWindows PRIVATE Threads::Threads)
    set(NEPTYPE_ENGINE_BINARY_DIR "" CACHE PATH "Directory containing NepTypeEngine.dll and its embedded dictionary")
    if(NOT EXISTS "${NEPTYPE_ENGINE_BINARY_DIR}/NepTypeEngine.dll")
        message(FATAL_ERROR "Build the shared Rust DLL with cargo build --release and set NEPTYPE_ENGINE_BINARY_DIR")
    endif()
    file(GLOB NEPTYPE_ENGINE_DLLS "${NEPTYPE_ENGINE_BINARY_DIR}/*.dll")
    foreach(ENGINE_DLL IN LISTS NEPTYPE_ENGINE_DLLS)
        add_custom_command(TARGET NepTypeWindows POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different "${ENGINE_DLL}" "$<TARGET_FILE_DIR:NepTypeWindows>")
    endforeach()
    add_custom_command(TARGET NepTypeWindows POST_BUILD
        COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/engine-data.py"
            stage "$<TARGET_FILE_DIR:NepTypeWindows>/dictionaries")

    if(MINGW)
        target_link_options(NepTypeWindows PRIVATE -municode)
    endif()
    # Build only; there is deliberately no installation or registration target.
    add_library(NepTypeTsf SHARED tsf/service.cpp tsf/edit_bridge.cpp tsf/owned_range.cpp tsf/exports.def)
    target_include_directories(NepTypeTsf PRIVATE tsf include)
    target_compile_features(NepTypeTsf PRIVATE cxx_std_17)
    target_link_libraries(NepTypeTsf PRIVATE NepTypeComposition ole32 uuid)
endif()
