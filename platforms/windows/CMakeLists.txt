cmake_minimum_required(VERSION 3.20)
project(LekhaWindowsPrototype LANGUAGES CXX)
add_library(LekhaComposition STATIC composition.cpp candidate_adapter.cpp)
target_include_directories(LekhaComposition PUBLIC include)
target_compile_features(LekhaComposition PUBLIC cxx_std_17)
include(CTest)
if(BUILD_TESTING)
    add_executable(LekhaCompositionTests tests/composition_test.cpp)
    target_link_libraries(LekhaCompositionTests PRIVATE LekhaComposition)
    add_test(NAME LekhaCompositionTests COMMAND LekhaCompositionTests)
    add_executable(LekhaCandidateTests tests/candidate_adapter_test.cpp)
    target_link_libraries(LekhaCandidateTests PRIVATE LekhaComposition)
    add_test(NAME LekhaCandidateTests COMMAND LekhaCandidateTests)
    add_executable(LekhaTsfLifecycleTests tsf/service.cpp tsf/edit_bridge.cpp tests/tsf_lifecycle_test.cpp)
    target_compile_features(LekhaTsfLifecycleTests PRIVATE cxx_std_17)
    target_compile_definitions(LekhaTsfLifecycleTests PRIVATE LEKHA_TSF_TEST)
    target_include_directories(LekhaTsfLifecycleTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(LekhaTsfLifecycleTests PRIVATE LekhaComposition)
    add_test(NAME LekhaTsfLifecycleTests COMMAND LekhaTsfLifecycleTests)
    add_executable(LekhaTsfEditTests tsf/edit_bridge.cpp tests/tsf_edit_test.cpp)
    target_compile_features(LekhaTsfEditTests PRIVATE cxx_std_17)
    target_compile_definitions(LekhaTsfEditTests PRIVATE LEKHA_TSF_TEST)
    target_include_directories(LekhaTsfEditTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(LekhaTsfEditTests PRIVATE LekhaComposition)
    add_test(NAME LekhaTsfEditTests COMMAND LekhaTsfEditTests)
    add_executable(LekhaOwnedRangeTests tsf/owned_range.cpp tests/owned_range_test.cpp)
    target_compile_features(LekhaOwnedRangeTests PRIVATE cxx_std_17)
    target_compile_definitions(LekhaOwnedRangeTests PRIVATE LEKHA_TSF_TEST)
    target_include_directories(LekhaOwnedRangeTests PRIVATE tests/fake_windows tsf include)
    target_link_libraries(LekhaOwnedRangeTests PRIVATE LekhaComposition)
    add_test(NAME LekhaOwnedRangeTests COMMAND LekhaOwnedRangeTests)
endif()
if(WIN32)
    add_executable(LekhaWindows WIN32 main.cpp)
    target_compile_features(LekhaWindows PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(LekhaWindows PRIVATE /utf-8)
    endif()
    target_link_libraries(LekhaWindows PRIVATE user32 gdi32 comctl32 LekhaComposition)
    if(MINGW)
        target_link_options(LekhaWindows PRIVATE -municode)
    endif()
    # Build only; there is deliberately no installation or registration target.
    add_library(LekhaTsf SHARED tsf/service.cpp tsf/edit_bridge.cpp tsf/owned_range.cpp)
    target_include_directories(LekhaTsf PRIVATE tsf include)
    target_compile_features(LekhaTsf PRIVATE cxx_std_17)
    target_link_libraries(LekhaTsf PRIVATE LekhaComposition msctf ole32 uuid)
endif()
