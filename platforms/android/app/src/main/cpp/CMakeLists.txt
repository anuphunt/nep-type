cmake_minimum_required(VERSION 3.22)
project(NepTypeJNI LANGUAGES CXX)
set(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../..")
set(ENGINE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../build/generated/shared-jni/${ANDROID_ABI}")
if(NOT EXISTS "${ENGINE_DIR}/libNepTypeEngine.so")
    message(FATAL_ERROR "Stage the shared Rust engine using scripts/build-android-engine.py before assembling Android")
endif()
add_library(NepTypeEngine SHARED IMPORTED)
set_target_properties(NepTypeEngine PROPERTIES IMPORTED_LOCATION "${ENGINE_DIR}/libNepTypeEngine.so")
add_library(neptype_jni SHARED "${REPO_ROOT}/shared/cpp/neptype_jni.cpp")
target_compile_features(neptype_jni PRIVATE cxx_std_17)
target_link_libraries(neptype_jni PRIVATE NepTypeEngine)
